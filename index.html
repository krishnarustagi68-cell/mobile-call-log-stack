 Course              : DIT University - CSN102 Data Structure
 Operations Supported :
   1. Add New Call Record (PUSH)
   2. Delete Recent Call (POP)
   3. View Last Dialed / Recent Call (PEEK / TOP)
   4. View Complete Call Log (TRAVERSAL - top to bottom)
   5. Check if Call Log is Empty (ISEMPTY)
   6. Exit
 ========================================================= */

#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX 50 /* Maximum number of call records the stack can hold */

/* ---------- Structure to store a single call record ---------- */
typedef struct {
    char number[15];    /* phone number */
    char type[10];      /* Incoming / Outgoing / Missed */
    int duration;       /* duration in seconds */
    char timestamp[20]; /* date-time of call, given as string */
} CallRecord;

/* ---------- Stack structure ---------- */
typedef struct {
    CallRecord calls[MAX];
    int top; /* index of the top element, -1 means empty */
} CallStack;

/* ---------- Function prototypes ---------- */
void initStack(CallStack *s);
int isFull(CallStack *s);
int isEmpty(CallStack *s);
void pushCall(CallStack *s, CallRecord c);
int popCall(CallStack *s, CallRecord *deleted);
int peekCall(CallStack *s, CallRecord *top);
void viewCallLog(CallStack *s);
void addNewCall(CallStack *s);
void menu(void);

/* ---------- Initialize the stack ---------- */
void initStack(CallStack *s) {
    s->top = -1;
}

/* ---------- Check if stack is full ---------- */
int isFull(CallStack *s) {
    return s->top == MAX - 1;
}

/* ---------- Check if stack is empty ---------- */
int isEmpty(CallStack *s) {
    return s->top == -1;
}

/* ---------- PUSH: Add a new call record on top of stack ---------- */
void pushCall(CallStack *s, CallRecord c) {
    if (isFull(s)) {
        printf("\nCall log is FULL! Cannot add more calls.\n");
        return;
    }
    s->top++;
    s->calls[s->top] = c;
    printf("\nCall record added successfully.\n");
}

/* ---------- POP: Delete the most recent call from the stack ---------- */
int popCall(CallStack *s, CallRecord *deleted) {
    if (isEmpty(s)) {
        return 0; /* failure - nothing to delete */
    }
    *deleted = s->calls[s->top];
    s->top--;
    return 1; /* success */
}

/* ---------- PEEK: View last dialed / most recent call without removing it ---------- */
int peekCall(CallStack *s, CallRecord *topCall) {
    if (isEmpty(s)) {
        return 0;
    }
    *topCall = s->calls[s->top];
    return 1;
}

/* ---------- Display the entire call log (most recent first) ---------- */
void viewCallLog(CallStack *s) {
    if (isEmpty(s)) {
        printf("\nCall log is empty. No calls to display.\n");
        return;
    }
    printf("\n--------------------------------------------------------------\n");
    printf("%-15s %-10s %-10s %-15s\n", "Number", "Type", "Duration", "Timestamp");
    printf("--------------------------------------------------------------\n");
    for (int i = s->top; i >= 0; i--) {
        printf("%-15s %-10s %-10d %-15s\n",
               s->calls[i].number, s->calls[i].type,
               s->calls[i].duration, s->calls[i].timestamp);
    }
    printf("--------------------------------------------------------------\n");
    printf("Total calls in log: %d\n", s->top + 1);
}

Why Stack (LIFO) is the Ideal Data Structure:

In modern smartphones, user interaction with call records inherently favors recency: checking the last dialed number or clearing the most recent accidental dial. By mapping operations to a Stack:

  • Add Call (PUSH): Places new record onto the top index in O(1) time.
  • Delete Most Recent Call (POP): Removes only the topmost entry in O(1) time without moving other elements.
  • View Last Dialed/Received Call (PEEK): Inspects calls[s->top] in O(1) time.
  • View Complete Call Log: Traverses top-to-bottom in O(n) time.
