#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MAX 50 /* Maximum capacity of the bounded call log stack */

typedef struct {
    char number[15];    /* Phone number */
    char type[10];      /* Incoming / Outgoing / Missed */
    int duration;       /* Duration in seconds */
    char timestamp[20]; /* Date-time string (DD-MM-YYYY) */
} CallRecord;

typedef struct {
    CallRecord calls[MAX];
    int top; /* Index of top element, -1 means empty */
} CallStack;

void initStack(CallStack *s) { s->top = -1; }
int isFull(CallStack *s) { return s->top == MAX - 1; }
int isEmpty(CallStack *s) { return s->top == -1; }

void pushCall(CallStack *s, CallRecord c) {
    if (isFull(s)) {
        printf("\nCall log is FULL! Cannot add more calls.\n");
        return;
    }
    s->top++;
    s->calls[s->top] = c;
    printf("\nCall record added successfully.\n");
}

int popCall(CallStack *s, CallRecord *deleted) {
    if (isEmpty(s)) return 0;
    *deleted = s->calls[s->top];
    s->top--;
    return 1;
}

int peekCall(CallStack *s, CallRecord *topCall) {
    if (isEmpty(s)) return 0;
    *topCall = s->calls[s->top];
    return 1;
}

void viewCallLog(CallStack *s) {
    if (isEmpty(s)) {
        printf("\nCall log is empty. No calls to display.\n");
        return;
    }
    printf("\n--------------------------------------------------------------\n");
    printf("%-15s %-10s %-10s %-15s\n", "Number", "Type", "Duration", "Timestamp");
    printf("--------------------------------------------------------------\n");
    for (int i = s->top; i >= 0; i--) {
        printf("%-15s %-10s %-10d %-15s\n",
               s->calls[i].number, s->calls[i].type,
               s->calls[i].duration, s->calls[i].timestamp);
    }
    printf("--------------------------------------------------------------\n");
    printf("Total calls in log: %d\n", s->top + 1);
}

1. Why Stack (LIFO) is the Optimal Choice

Mobile call history is inherently governed by Last-In-First-Out (LIFO) recency. When dialing or clearing calls, users interact with the most recent entry first. Using a Stack allows PUSH, POP, and PEEK in constant O(1) time.

2. Array vs Linked-List Stack Analysis

An Array-based stack provides contiguous memory allocation, zero pointer overhead (saving 8 bytes per node on 64-bit systems), and maximum CPU cache locality. The bounded capacity MAX = 50 mirrors real mobile logs.

3. Asymptotic Time & Space Proof

Push: O(1) | Pop: O(1) | Peek: O(1) | Traverse: O(n)
Space Complexity: O(MAX) fixed at compile-time (2,404 bytes), preventing dynamic heap fragmentation.

4. Error & Boundary Handling

• Underflow: isEmpty(s) guards pop and peek when top == -1.
• Overflow: isFull(s) guards push when top == MAX - 1, preventing buffer overflows.
